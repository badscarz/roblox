-- We assume these variables (proFrame, h1iyFrame, walkFrame) are defined above this
local menuItems = {proFrame, h1iyFrame, walkFrame}
local currentIndex = 1

-- 2. Visual Feedback Function
-- Highlights the selected menu and dims the others
local function updateSelection()
    for i, frame in ipairs(menuItems) do
        if i == currentIndex then
            frame.BackgroundTransparency = 0.1 -- Highlighted
            frame.BorderSizePixel = 2
            frame.BorderColor3 = Color3.fromRGB(255, 255, 255) -- White glow
        else
            frame.BackgroundTransparency = 0.5 -- Dimmed
            frame.BorderSizePixel = 0
        end
    end
end

-- 3. Navigation Logic
UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    
    -- Arrow Down: Move to next item (with Loop)
    if input.KeyCode == Enum.KeyCode.Down then
        currentIndex = currentIndex + 1
        if currentIndex > #menuItems then
            currentIndex = 1 -- Loop back to start
        end
        updateSelection()
        
    -- Arrow Up: Move to previous item (with Loop)
    elseif input.KeyCode == Enum.KeyCode.Up then
        currentIndex = currentIndex - 1
        if currentIndex < 1 then
            currentIndex = #menuItems -- Loop to end
        end
        updateSelection()
        
    -- Enter: Toggle the main feature of the focused menu
    elseif input.KeyCode == Enum.KeyCode.Return then
        local focused = menuItems[currentIndex]
        -- Logic to trigger the 'Primary' button of that specific frame
        if focused == proFrame then
            print("Activating Hybrid Pro Features...")
        elseif focused == h1iyFrame then
            print("Activating h1iy Controller...")
        elseif focused == walkFrame then
            -- This triggers the Auto-Forward logic we built earlier
            toggleForward() 
        end
        
    -- Backspace: Reset focus to the first item
    elseif input.KeyCode == Enum.KeyCode.Backspace then
        currentIndex = 1
        updateSelection()
    end
end)

-- Initialize the visual state
updateSelection()

local function makeDraggable(frame)
	local dragging, dragInput, dragStart, startPos
	local function update(input)
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
			startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
	frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging=false end
			end)
		end
	end)
	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement then dragInput = input end
	end)
	RunService.RenderStepped:Connect(function()
		if dragging and dragInput then update(dragInput) end
	end)
end

local function isInLineOfSight(target)
	local targetPos = target.Character[aimlockData.TargetPart].Position
	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = {character}
	rayParams.FilterType = Enum.RaycastFilterType.Blacklist
	local ray = Workspace:Raycast(root.Position, (targetPos - root.Position), rayParams)
	return not ray or ray.Instance:IsDescendantOf(target.Character)
end

local function getNearestPlayer()
	local nearest, minDist
	for _, p in ipairs(Players:GetPlayers()) do
		if p~=player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
			local dist = (p.Character.HumanoidRootPart.Position - root.Position).Magnitude
			if isInLineOfSight(p) then
				if not minDist or dist<minDist then
					minDist = dist
					nearest = p
				end
			end
		end
	end
	return nearest
end

local function avoidObstacles(desiredDir)
	local params = RaycastParams.new()
	params.FilterDescendantsInstances={character}
	params.FilterType=Enum.RaycastFilterType.Blacklist
	local hit = Workspace:Raycast(root.Position, desiredDir*OBSTACLE_AVOID_RADIUS, params)
	if hit and hit.Instance and hit.Instance.CanCollide then
		local normal = hit.Normal
		desiredDir = (desiredDir + normal*1.2).Unit
	end
	return desiredDir
end

local function momentumMoveToward(targetPos, dt)
	local dir = targetPos - root.Position
	dir = Vector3.new(dir.X,0,dir.Z)
	if dir.Magnitude<0.5 then velocity *= MOMENTUM_DECAY else
		local desiredDir = avoidObstacles(dir.Unit)
		velocity = velocity:Lerp(desiredDir*humanoid.WalkSpeed,MOMENTUM_GAIN)
	end
	if velocity.Magnitude<0.1 then velocity=Vector3.zero end
	if velocity.Magnitude>0 then humanoid:Move(velocity.Unit*math.clamp(velocity.Magnitude/humanoid.WalkSpeed,0,1),false) end
end

end

-- Stuck Detection Loop
task.spawn(function()
	while true do
		task.wait(STUCK_CHECK_INTERVAL)
		if root then
			table.insert(lastPositions,root.Position)
			if #lastPositions>math.ceil(STUCK_TIME/STUCK_CHECK_INTERVAL) then table.remove(lastPositions,1) end
			if #lastPositions>=2 then
				local first = lastPositions[1]
				local maxDist=0
				for _,pos in ipairs(lastPositions) do
					local dist=(pos-first).Magnitude
					if dist>maxDist then maxDist=dist end
				end
				isStuck = maxDist<STUCK_RADIUS
			end
		end
	end
end)

-- Aimlock Target Finding
local function getAimlockTarget()
	local closest,minAngle = nil,aimlockData.FOV
	for _,p in ipairs(Players:GetPlayers()) do
		if p~=player and p.Character and p.Character:FindFirstChild(aimlockData.TargetPart) then
			local partPos = p.Character[aimlockData.TargetPart].Position
			local screenPos = camera:WorldToViewportPoint(partPos)
			local direction = (Vector2.new(screenPos.X,screenPos.Y)-Vector2.new(camera.ViewportSize.X/2,camera.ViewportSize.Y/2)).Magnitude
			if direction<minAngle then
				local rayParams = RaycastParams.new()
				rayParams.FilterDescendantsInstances={character}
				rayParams.FilterType=Enum.RaycastFilterType.Blacklist
				local ray = Workspace:Raycast(camera.CFrame.Position, (partPos-camera.CFrame.Position), rayParams)
				if not ray or ray.Instance:IsDescendantOf(p.Character) then
					closest = p
					minAngle = direction
				end
			end
		end
	end
	return closest
end

local function aimAtTarget(target, dt)
	if target and target.Character and target.Character:FindFirstChild(aimlockData.TargetPart) then
		local partPos = target.Character[aimlockData.TargetPart].Position
		local velocityPredict = target.Character:FindFirstChild("HumanoidRootPart") and target.Character.HumanoidRootPart.Velocity or Vector3.zero
		local predictedPos = partPos + velocityPredict*aimlockData.Prediction
		local camPos = camera.CFrame.Position
		local direction = predictedPos - camPos
		local offset = infinityPattern[math.floor(infinityIndex)]*0.5
		infinityIndex = infinityIndex + INFINITY_SPEED*dt
		if infinityIndex>#infinityPattern then infinityIndex=1 end
		local camDir = Vector3.new(direction.Unit.X*CAMERA_X_SENS+offset.X, direction.Unit.Y*CAMERA_Y_SENS+offset.Y, direction.Unit.Z)
		camera.CFrame = camera.CFrame:Lerp(CFrame.new(camPos, camPos+camDir), CAMERA_LERP_SPEED*60*dt)
	end
end

	currentTarget = getNearestPlayer()
	if currentTarget then status.Text="Target: "..currentTarget.Name else status.Text="Target: None" end

	if currentTarget and currentTarget.Character then
		local targetPos = getTargetPosition(currentTarget.Character)
		if targetPos and (glideEnabled or autoWalkEnabled) then
			momentumMoveToward(targetPos, dt)
			rotateCameraToward(targetPos)
		end
	end

	if (autoWalkEnabled or glideEnabled) and isStuck then
		lastStuckMove += dt
		if lastStuckMove>=STUCK_MOVE_TIME then lastStuckMove=0; stuckIndex=stuckIndex%#stuckStrafeDirs+1 end
		humanoid:Move(stuckStrafeDirs[stuckIndex],false)
	end

	if jumpMovementEnabled and currentTarget and currentTarget.Character then
		local targetHum = currentTarget.Character:FindFirstChild("Humanoid")
		if targetHum and targetHum.FloorMaterial==Enum.Material.Air and humanoid.FloorMaterial~=Enum.Material.Air then
			humanoid.Jump=true
		end
	end

	if (traceEnabled or autoShootEnabled) and currentTarget and currentTarget.Character then
		local targetPos = getTargetPosition(currentTarget.Character)
		if targetPos and (targetPos-root.Position).Magnitude <= TRACE_FIRE_DISTANCE then
			local tool = character:FindFirstChildOfClass("Tool")
			if tool then tool:Activate() end
		end
	end

	if aimlockEnabled then aimAtTarget(getAimlockTarget(), dt) end

	if aimbotEnabled and currentTarget and currentTarget.Character:FindFirstChild(TARGET_PART) then
		camera.CFrame = CFrame.new(camera.CFrame.Position, currentTarget.Character[TARGET_PART].Position)
	end

	for p, esp in pairs(espObjects) do
		if p.Character and p.Character:FindFirstChild("HumanoidRootPart") and SHOW_ESP then
			local rootPos, onScreen = camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
			if onScreen then
				esp.box.Size = Vector2.new(1000/rootPos.Z, 1000/rootPos.Z * 1.5)
				esp.box.Position = Vector2.new(rootPos.X - esp.box.Size.X/2, rootPos.Y - esp.box.Size.Y/2)
				esp.box.Visible = true
			else esp.box.Visible = false end
		else esp.box.Visible = false end
	end
end)
